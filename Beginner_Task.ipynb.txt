        "# 4.3 (to check if two belongs to a country or not)\n",
        "import requests\n",
        "\n",
        "def get_country(city):\n",
        "    url = \"https://nominatim.openstreetmap.org/search\"\n",
        "    params = {\n",
        "        \"q\": city,\n",
        "        \"format\": \"json\",\n",
        "        \"addressdetails\": 1,\n",
        "        \"limit\": 1\n",
        "    }\n",
        "    headers = {\n",
        "        \"User-Agent\": \"city-country-checker-app\"  # required by Nominatim's usage policy\n",
        "    }\n",
        "\n",
        "    response = requests.get(url, params=params, headers=headers)\n",
        "    data = response.json()\n",
        "\n",
        "    if not data:\n",
        "        return None, None\n",
        "\n",
        "    address = data[0].get(\"address\", {})\n",
        "    country = address.get(\"country\")\n",
        "    display_name = data[0].get(\"display_name\")\n",
        "    return country, display_name\n",
        "\n",
        "\n",
        "def main():\n",
        "    city1 = input(\"Enter the first city: \").strip()\n",
        "    city2 = input(\"Enter the second city: \").strip()\n",
        "\n",
        "    country1, name1 = get_country(city1)\n",
        "    country2, name2 = get_country(city2)\n",
        "\n",
        "    if country1 is None:\n",
        "        print(f\"Couldn't find location data for '{city1}'.\")\n",
        "        return\n",
        "    if country2 is None:\n",
        "        print(f\"Couldn't find location data for '{city2}'.\")\n",
        "        return\n",
        "\n",
        "    print(f\"\\n{city1} → {name1}\")\n",
        "    print(f\"{city2} → {name2}\\n\")\n",
        "\n",
        "    if country1.lower() == country2.lower():\n",
        "        print(f\"✅ Both cities are in the same country: {country1}\")\n",
        "    else:\n",
        "        print(f\"❌ Different countries: {country1} vs {country2}\")\n",

        "def roll_dice_simulation(num_rolls=20):\n",

        "\n",
        "    count_6 = 0\n",
        "    count_1 = 0\n",
        "    double_sixes = 0\n",
        "\n",
        "    for i in range(len(rolls)):\n",
        "        if rolls[i] == 6:\n",
        "            count_6 += 1\n",
        "        if rolls[i] == 1:\n",
        "            count_1 += 1\n",
        "        if rolls[i] == 6 and i > 0 and rolls[i - 1] == 6:\n",
        "            double_sixes += 1\n",
