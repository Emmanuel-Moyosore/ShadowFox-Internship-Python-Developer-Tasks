        "import csv\n",
        "import sys\n",
        "import time\n",
        "\n",
        "try:\n",
        "    import requests\n",
        "    from bs4 import BeautifulSoup\n",

        "    print(\"Missing dependencies. Install them with:\")\n",
        "    print(\"    pip install requests beautifulsoup4\")\n",
        "    sys.exit(1)\n",
        "\n",
        "\n",
        "HEADERS = {\n",
        "    \"User-Agent\": (\n",
        "        \"Mozilla/5.0 (Windows NT 10.0; Win64; x64) \"\n",
        "        \"AppleWebKit/537.36 (KHTML, like Gecko) \"\n",
        "        \"Chrome/120.0 Safari/537.36\"\n",
        "    )\n",
        "}\n",
        "\n",
        "\n",
        "def fetch_page(url, retries=3, delay=2):\n",
        "    \"\"\"Fetch a page's HTML with basic retry/error handling.\"\"\"\n",
        "    for attempt in range(1, retries + 1):\n",
        "        try:\n",
        "            response = requests.get(url, headers=HEADERS, timeout=10)\n",
        "            response.raise_for_status()\n",
        "            return response.text\n",
        "        except requests.exceptions.RequestException as e:\n",
        "            print(f\"Attempt {attempt}/{retries} failed: {e}\")\n",
        "            if attempt < retries:\n",
        "                time.sleep(delay)\n",
        "    print(f\"Could not fetch {url} after {retries} attempts.\")\n",
        "    return None\n",
        "\n",
        "\n",
        "def parse_links_and_headings(html, base_url):\n",
        "    \"\"\"Extract all headings (h1-h3) and links from the page.\"\"\"\n",
        "    soup = BeautifulSoup(html, \"html.parser\")\n",
        "\n",
        "    headings = []\n",
        "    for level in (\"h1\", \"h2\", \"h3\"):\n",
        "        for tag in soup.find_all(level):\n",
        "            text = tag.get_text(strip=True)\n",
        "            if text:\n",
        "                headings.append({\"level\": level, \"text\": text})\n",
        "\n",
        "    links = []\n",
        "    for a_tag in soup.find_all(\"a\", href=True):\n",
        "        text = a_tag.get_text(strip=True)\n",
        "        href = a_tag[\"href\"]\n",
        "        # Resolve relative links into absolute URLs\n",
        "        if href.startswith(\"/\"):\n",
        "            href = base_url.rstrip(\"/\") + href\n",
        "        links.append({\"text\": text or \"(no text)\", \"url\": href})\n",
        "\n",
        "    return headings, links\n",
        "\n",
        "\n",
        "def save_to_csv(rows, fieldnames, filename):\n",
        "    if not rows:\n",
        "        print(f\"Nothing to save for {filename}.\")\n",
        "        return\n",
        "    try:\n",
        "        with open(filename, \"w\", newline=\"\", encoding=\"utf-8\") as f:\n",
        "            writer = csv.DictWriter(f, fieldnames=fieldnames)\n",
        "            writer.writeheader()\n",
        "            writer.writerows(rows)\n",
        "        print(f\"Saved {len(rows)} rows to {filename}\")\n",
        "    except OSError as e:\n",
        "        print(f\"Error writing {filename}: {e}\")\n",
        "\n",
        "\n",
